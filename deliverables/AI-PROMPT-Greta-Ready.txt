  <script>
    const API = 'https://salah.chargedesk.de';
    const CITY = 'Offenbach';
    const today = new Date();
    const isoDate = today.toISOString().slice(0, 10);
    const startDate = isoDate;
    const lang = navigator.language || 'de';

    // 1) Today's prayer times
    fetch(API + '/api/times?city=' + encodeURIComponent(CITY) + '&date=' + isoDate)
      .then(r => r.json())
      .then(d => {
        document.getElementById('city-name').textContent = '🕌 ' + d.city.name;
        const dateStr = new Date(d.date).toLocaleDateString(lang, { weekday:'long', day:'2-digit', month:'long', year:'numeric' });
        document.getElementById('date-display').textContent = dateStr + ' · ' + d.city.timezone;

        // Find next prayer
        const order = ['imsak','sunrise','dhuhr','asr','maghrib','isha'];
        const labels = { imsak:'İmsak', sunrise:'Güneş', dhuhr:'Öğle', asr:'İkindi', maghrib:'Akşam', isha:'Yatsı' };
        const now = new Date();
        const mins = now.getHours()*60 + now.getMinutes();
        let next = null;
        for (const k of order) {
          const [h,m] = d[k].split(':').map(Number);
          if (h*60+m > mins) { next = { key:k, time:d[k] }; break; }
        }
        if (next) {
          document.getElementById('next-pill').innerHTML = '<span class="next-pill">Sıradaki: ' + labels[next.key] + ' ' + next.time + '</span>';
        }

        // Render today's 6 prayers
        const grid = document.getElementById('prayer-grid');
        grid.innerHTML = order.map(k =>
          '<div class="prayer-cell' + (next && next.key===k ? ' current' : '') + '">' +
            '<div class="label">' + labels[k] + '</div>' +
            '<div class="time">' + d[k] + '</div>' +
          '</div>'
        ).join('');
      })
      .catch(e => {
        document.getElementById('prayer-grid').innerHTML = '<div class="error">❌ ' + e.message + '</div>';
      });

    // 2) Week overview
    fetch(API + '/api/times/range?city=' + encodeURIComponent(CITY) + '&start=' + startDate + '&days=7')
      .then(r => r.json())
      .then(d => {
        const labels = { imsak:'İmsak', sunrise:'Güneş', dhuhr:'Öğle', asr:'İkindi', maghrib:'Akşam', isha:'Yatsı' };
        const body = document.getElementById('week-body');
        body.innerHTML = d.times.map(t => {
          const ds = new Date(t.date + 'T00:00:00');
          const isToday = t.date === isoDate;
          const dayName = ds.toLocaleDateString(lang, { weekday:'short', day:'2-digit', month:'2-digit' });
          return '<tr class="' + (isToday?'today':'') + '"><td><strong>' + dayName + '</strong></td>' +
            '<td>' + t.imsak + '</td><td>' + t.sunrise + '</td>' +
            '<td>' + t.dhuhr + '</td><td>' + t.asr + '</td>' +
            '<td>' + t.maghrib + '</td><td>' + t.isha + '</td></tr>';
        }).join('');
        document.getElementById('week-table').style.display = 'table';
      });
  </script>
</body>

=== API ENDPOINTS (für weitere Entwicklung) ===
- Heute:    https://salah.chargedesk.de/api/times?city=Offenbach&date=YYYY-MM-DD
- 7 Tage:   https://salah.chargedesk.de/api/times/range?city=Offenbach&start=YYYY-MM-DD&days=7
- Monat:    https://salah.chargedesk.de/api/times/month?city=Offenbach&year=2026&month=8
- Widget:   <iframe src="https://salah.chargedesk.de/widget?city=Offenbach" width="100%" height="380"></iframe>
- Auto-Create: https://salah.chargedesk.de/api/cities/custom (POST) - fügt eine neue Stadt hinzu
